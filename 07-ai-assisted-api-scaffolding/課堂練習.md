## 要審查的程式碼

**`src/routes/contacts.routes.js`**

```js
import { Router } from 'express'
import * as contactsController from '../controllers/contacts.controller.js'

export const contactsRouter = Router()

contactsRouter.get('/', contactsController.list)
contactsRouter.get('/:id', contactsController.get)
contactsRouter.post('/', contactsController.create)
contactsRouter.put('/:id', contactsController.update)
contactsRouter.post('/:id/delete', contactsController.remove)
```

**`src/controllers/contacts.controller.js`**

```js
import * as contactsService from '../services/contacts.service.js'

export async function list(req, res) {
  res.json(await contactsService.listContacts())
}

export async function get(req, res) {
  const contact = await contactsService.getContact(req.params.id)
  if (!contact) {
    return res.status(404).json({ message: 'Contact not found' })
  }
  res.json(contact)
}

export async function create(req, res) {
  try {
    const { name, email } = req.body
    if (!name || !email) {
      return res.status(400).json({ error: 'name 和 email 必填' })
    }
    const contact = await contactsService.createContact({ name, email })
    res.status(200).json(contact)
  } catch (err) {
    res.status(500).json({ error: err.message })
  }
}

export async function update(req, res) {
  const contact = await contactsService.updateContact(Number(req.params.id), req.body)
  res.json(contact)
}

export async function remove(req, res) {
  await contactsService.deleteContact(Number(req.params.id), res)
}
```

**`src/services/contacts.service.js`**

```js
import { HttpError } from '../lib/httpError.js'

const contacts = []

export async function listContacts() {
  return contacts
}

export async function getContact(id) {
  return contacts.find((c) => c.id === id)
}

export async function createContact({ name, email }) {
  const contact = { id: contacts.length + 1, name, email }
  contacts.push(contact)
  return contact
}

export async function updateContact(id, data) {
  if (data.email !== undefined && !data.email.includes('@')) {
    throw new HttpError(400, 'email 格式錯誤')
  }
  const contact = contacts.find((c) => c.id === id)
  if (!contact) throw new HttpError(404, '找不到這個聯絡人')
  Object.assign(contact, data)
  return contact
}

export async function deleteContact(id, res) {
  const index = contacts.findIndex((c) => c.id === id)
  if (index === -1) throw new HttpError(404, '找不到這個聯絡人')
  contacts.splice(index, 1)
  res.status(200).json({ message: 'deleted' })
}
```

**明確規格**（這幾條就是這堂重點，別漏）：

- 每一列都要寫出**規則編號**。找到的問題如果規則檔裡沒有對應的規則，在「違反哪一條規則」寫「規則檔沒寫到」，並在最後提出一條新規則。
- 「實際會造成什麼問題」要具體，例如「前端要多處理一種錯誤格式」，不能只寫「不符規範」。
- 至少有一個問題要**實際跑起來驗證**：把三個檔案放進第 7 堂完成版、在 `routes/index.js` 掛上 `/contacts`，用 curl 重現它。
- 不要直接請 AI 幫你審查這份程式碼。這個練習要練的是**你自己**的判斷。
